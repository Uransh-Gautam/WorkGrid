-- 1. SQL Query:
SELECT e.id,e.name,40-COALESCE(SUM(a.hours_allocated),0) hours_available
FROM employee e LEFT JOIN allocation a ON a.employee_id=e.id AND a.status='active'
WHERE e.id=6 GROUP BY e.id,e.name;
